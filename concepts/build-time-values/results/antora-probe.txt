# make probe-dates (Antora): Asciidoctor's date attributes and the sitemap lastmod, without and with SOURCE_DATE_EPOCH
mkdir -p .check
cp probes/dates.adoc pages/modules/ROOT/pages/probe-dates.adoc
sites/antora/node_modules/.bin/antora --stacktrace antora-playbook.yml --to-dir .check/probe-site > /dev/null 2>&1; \
	  grep -o '<p>[a-z]*: [^<]*</p>' .check/probe-site/cf/probe-dates/index.html > .check/probe-dates.log; \
	  echo "sitemap lastmod: $(grep -o '<lastmod>[^<]*' .check/probe-site/sitemap.xml | sort -u | tr '\n' ' ')" >> .check/probe-dates.log
echo '## SOURCE_DATE_EPOCH=1733437108 (2024-12-05T22:18:28Z)' >> .check/probe-dates.log
SOURCE_DATE_EPOCH=1733437108 sites/antora/node_modules/.bin/antora --stacktrace antora-playbook.yml --to-dir .check/probe-site > /dev/null 2>&1; \

	  echo "sitemap lastmod: $(grep -o '<lastmod>[^<]*' .check/probe-site/sitemap.xml | sort -u | tr '\n' ' ')" >> .check/probe-dates.log
